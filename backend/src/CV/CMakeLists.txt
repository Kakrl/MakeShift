cmake_minimum_required(VERSION 3.15)
project(MakeShiftCV LANGUAGES CXX)

# Standalone experiment: no PortAudio, nanobind, or browser runtime.
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MAKESHIFT_LIVE_CAMERA "Build the optional desktop camera viewer" OFF)
set(cv_components core imgproc imgcodecs)
if(MAKESHIFT_LIVE_CAMERA)
    list(APPEND cv_components highgui videoio objdetect)
endif()
find_package(OpenCV 4.12 CONFIG REQUIRED COMPONENTS ${cv_components})

add_library(makeshift_cv_baseline STATIC Homography.cpp ShadowDetection.cpp Fingertips.cpp CrFingerDetection.cpp ShadowPaths.cpp LightShadow.cpp)
target_include_directories(makeshift_cv_baseline PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(makeshift_cv_baseline PUBLIC opencv_core)
add_executable(shadow_images shadow_images.cpp)
add_library(makeshift_cv_diagnostics STATIC ShadowDiagnostics.cpp ColorDiagnostics.cpp)
target_link_libraries(makeshift_cv_diagnostics PUBLIC makeshift_cv_baseline opencv_imgproc)
target_link_libraries(shadow_images PRIVATE makeshift_cv_diagnostics opencv_imgcodecs)
if(MAKESHIFT_LIVE_CAMERA)
    add_library(makeshift_camera_keyboard STATIC CameraKeyboard.cpp CameraSnapshot.cpp)
    target_link_libraries(makeshift_camera_keyboard PUBLIC makeshift_cv_baseline opencv_imgproc opencv_objdetect opencv_imgcodecs)
    add_executable(shadow_camera shadow_camera.cpp)
    target_link_libraries(shadow_camera PRIVATE makeshift_camera_keyboard opencv_highgui opencv_videoio opencv_imgcodecs)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(baseline_tests ../../../tests/cv/test_baseline.cpp)
    target_link_libraries(baseline_tests PRIVATE makeshift_cv_baseline opencv_imgproc opencv_imgcodecs)
    add_test(NAME cv_baseline COMMAND baseline_tests)
    if(MAKESHIFT_LIVE_CAMERA)
        add_executable(camera_tests ../../../tests/cv/test_camera.cpp)
        target_link_libraries(camera_tests PRIVATE makeshift_camera_keyboard)
        add_test(NAME cv_camera_keyboard_snapshot COMMAND camera_tests)
        add_test(NAME cv_camera_help COMMAND shadow_camera --help)
        set_tests_properties(cv_camera_help PROPERTIES PASS_REGULAR_EXPRESSION "capture empty sheet")
        add_test(NAME cv_camera_invalid_size COMMAND shadow_camera --max-side=641)
        set_tests_properties(cv_camera_invalid_size PROPERTIES WILL_FAIL TRUE)
        add_test(NAME cv_camera_invalid_device COMMAND shadow_camera --camera=-1)
        set_tests_properties(cv_camera_invalid_device PROPERTIES WILL_FAIL TRUE)
        add_test(NAME cv_camera_invalid_octaves COMMAND shadow_camera --octaves=4)
        set_tests_properties(cv_camera_invalid_octaves PROPERTIES WILL_FAIL TRUE)
    endif()
endif()
