cmake_minimum_required(VERSION 3.15)
project(MakeShiftVision LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MAKESHIFT_LIVE_CAMERA "Build the native live camera viewer" ON)
set(cv_components core imgproc)
if(MAKESHIFT_LIVE_CAMERA)
    list(APPEND cv_components objdetect highgui videoio)
endif()
find_package(OpenCV 4.12 CONFIG REQUIRED COMPONENTS ${cv_components})

add_library(makeshift_vision STATIC
    Pipeline.cpp ContactDetection.cpp Homography.cpp KeyboardGeometry.cpp FingerSegmentation.cpp
    Fingertips.cpp ShadowMeasurements.cpp DarkShadow.cpp LightShadow.cpp MidShadow.cpp)
target_include_directories(makeshift_vision PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(makeshift_vision PUBLIC opencv_core opencv_imgproc)
add_library(makeshift_shadow_view STATIC ShadowView.cpp)
target_link_libraries(makeshift_shadow_view PUBLIC makeshift_vision)
if(MAKESHIFT_LIVE_CAMERA)
    add_executable(shadow_camera shadow_camera.cpp MarkerDetection.cpp)
    target_link_libraries(shadow_camera PRIVATE makeshift_shadow_view opencv_objdetect opencv_highgui opencv_videoio)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(vision_tests ../../../tests/cv/test_pipeline.cpp)
    target_link_libraries(vision_tests PRIVATE makeshift_shadow_view)
    add_test(NAME cv_pipeline COMMAND vision_tests)
    if(MAKESHIFT_LIVE_CAMERA)
        add_test(NAME cv_camera_help COMMAND shadow_camera --help)
        set_tests_properties(cv_camera_help PROPERTIES PASS_REGULAR_EXPRESSION "Live video on left")
        foreach(arg IN ITEMS "--camera=-1" "--camera=1x" "--octaves=0" "--octaves=4" "--unknown")
            string(MAKE_C_IDENTIFIER "${arg}" test_id)
            add_test(NAME cv_reject_${test_id} COMMAND shadow_camera "${arg}")
            set_tests_properties(cv_reject_${test_id} PROPERTIES WILL_FAIL TRUE)
        endforeach()
    endif()
endif()
