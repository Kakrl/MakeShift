cmake_minimum_required(VERSION 3.15)
project(MakeShiftVision LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MAKESHIFT_LIVE_CAMERA "Build the native live camera viewer" ON)
if(EMSCRIPTEN)
    set(MAKESHIFT_LIVE_CAMERA OFF CACHE BOOL "No camera/window code in WASM" FORCE)
    add_compile_options(-fexceptions)
    add_link_options(-fexceptions -sALLOW_MEMORY_GROWTH=1)
endif()
set(cv_components core imgproc objdetect)
if(MAKESHIFT_LIVE_CAMERA)
    list(APPEND cv_components highgui videoio imgcodecs)
endif()
find_package(OpenCV 4.12 CONFIG REQUIRED COMPONENTS ${cv_components})

add_library(makeshift_vision STATIC
    Vision.cpp ChromaticLookup.cpp MarkerDetection.cpp Pipeline.cpp ContactDetection.cpp Homography.cpp KeyboardGeometry.cpp FingerSegmentation.cpp
    Fingertips.cpp ShadowMeasurements.cpp DarkShadow.cpp LightShadow.cpp MidShadow.cpp)
target_include_directories(makeshift_vision PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(makeshift_vision PUBLIC opencv_core opencv_imgproc opencv_objdetect)
add_library(makeshift_shadow_view STATIC ShadowView.cpp)
target_link_libraries(makeshift_shadow_view PUBLIC makeshift_vision)
if(MAKESHIFT_LIVE_CAMERA)
    add_executable(shadow_camera shadow_camera.cpp)
    target_link_libraries(shadow_camera PRIVATE makeshift_shadow_view opencv_objdetect opencv_highgui opencv_videoio opencv_imgcodecs)
    target_compile_definitions(shadow_camera PRIVATE MAKESHIFT_CASES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/cases")
endif()
if(EMSCRIPTEN)
    add_executable(vision_wasm wasm/Bindings.cpp)
    target_link_libraries(vision_wasm PRIVATE makeshift_vision)
    target_link_options(vision_wasm PRIVATE --bind -sMODULARIZE=1 -sEXPORT_ES6=1
        -sENVIRONMENT=web,worker,node -sFILESYSTEM=0 -sINITIAL_MEMORY=134217728
        -sMAXIMUM_MEMORY=1073741824)
    set_target_properties(vision_wasm PROPERTIES OUTPUT_NAME "makeshift-vision-core"
        SUFFIX ".mjs" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/dist")
    configure_file(wasm/makeshift-vision.mjs dist/makeshift-vision.mjs COPYONLY)
    configure_file(wasm/makeshift-vision.d.mts dist/makeshift-vision.d.mts COPYONLY)
    configure_file(wasm/toolchain.env dist/toolchain.env COPYONLY)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(chromatic_lookup_tests ../../../tests/cv/test_chromatic_lookup.cpp)
    target_link_libraries(chromatic_lookup_tests PRIVATE makeshift_vision)
    add_test(NAME cv_chromatic_lookup COMMAND chromatic_lookup_tests)
    add_executable(vision_tests ../../../tests/cv/test_pipeline.cpp)
    target_link_libraries(vision_tests PRIVATE makeshift_shadow_view)
    add_test(NAME cv_pipeline COMMAND vision_tests)
    add_executable(reference_policy_tests ../../../tests/cv/test_reference_policy.cpp)
    target_link_libraries(reference_policy_tests PRIVATE makeshift_vision)
    add_test(NAME cv_reference_policy COMMAND reference_policy_tests)
    if(NOT EMSCRIPTEN)
        add_executable(wasm_parity_native ../../../tests/cv/wasm_parity.cpp)
        target_link_libraries(wasm_parity_native PRIVATE makeshift_vision)
    endif()
    if(MAKESHIFT_LIVE_CAMERA)
        add_test(NAME cv_camera_help COMMAND shadow_camera --help)
        set_tests_properties(cv_camera_help PROPERTIES PASS_REGULAR_EXPRESSION "Live video on left")
        foreach(arg IN ITEMS "--camera=-1" "--camera=1x" "--octaves=0" "--octaves=4" "--unknown")
            string(MAKE_C_IDENTIFIER "${arg}" test_id)
            add_test(NAME cv_reject_${test_id} COMMAND shadow_camera "${arg}")
            set_tests_properties(cv_reject_${test_id} PROPERTIES WILL_FAIL TRUE)
        endforeach()
    endif()
endif()
